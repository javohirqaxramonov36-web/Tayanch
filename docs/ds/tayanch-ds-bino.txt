$("#fi-grads").innerHTML = FLOORS.map(f => {
    const c = `var(--color-floor-${f.key}-500)`, W = "var(--color-neutral-0)", K = "var(--color-neutral-950)";

function heroSVG(active) {
  const cxm = 130, a = 100, b = 50, h = 36, pitch = 44, base = 330, P = pts => pts.map(p => p.join(",")).join(" ");
  const y1 = base - h;
  let s = `<polygon class="shadow" filter="url(#fi-soft)" points="${P([[cxm - a + 8, y1 - b + h + 12], [cxm + 8, y1 + h + 12], [cxm + a + 8, y1 - b + h + 12], [cxm + 8, y1 - 2 * b + h + 12]])}"/>`;
  FLOORS.forEach(f => {
    const yF = base - (f.n - 1) * pitch - h;
    const L = [cxm - a, yF - b], F = [cxm, yF], R = [cxm + a, yF - b], Bk = [cxm, yF - 2 * b], Lb = [cxm - a, yF - b + h], Fb = [cxm, yF + h], Rb = [cxm + a, yF - b + h];
    const left = P([L, F, Fb, Lb]), right = P([F, R, Rb, Fb]), top = P([L, F, R, Bk]), sil = P([L, Bk, R, Rb, Fb, Lb]);
    const wins = [0, 1, 2, 3].map(k => `<rect class="win" style="--i:${k}" x="${16 + k * 21}" y="0" width="12" height="16" rx="1.5"/>`).join("");
    s += `<g class="b-floor" data-floor="${f.n}" data-on="${f.n === active}">
      <polygon class="glow" points="${sil}" filter="url(#fi-blur)"/>
      <polygon class="g-t" points="${top}"/><polygon class="g-l" points="${left}"/><polygon class="g-r" points="${right}"/>
      <g class="on"><polygon points="${top}" fill="url(#gT${f.n})"/><polygon points="${left}" fill="url(#gL${f.n})"/><polygon points="${right}" fill="url(#gR${f.n})"/></g>
      <polygon class="g-s" points="${top}"/><polygon class="g-s" points="${left}"/><polygon class="g-s" points="${right}"/>
      <polyline class="edge" points="${P([Lb, Fb, Rb])}"/>
      <g class="on"><polyline class="hl" points="${P([L, F, R])}"/></g>
      <g transform="translate(${cxm - a + 14},${yF - b + 30}) matrix(1 .5 0 1 0 0)" class="lab"><text x="0" y="0">${pad(f.n)}</text><use class="ic-stroke" href="#i-${f.icon}" x="25" y="-14" width="18" height="18"/></g>
      <g transform="translate(${cxm},${yF + 10}) matrix(1 -.5 0 1 0 0)">${wins}</g></g>`;
  });
  const y5 = base - 4 * pitch - h;
  s += `<line class="ant" x1="${cxm}" y1="${y5 - b}" x2="${cxm}" y2="${y5 - b - 34}"/><g class="tipg" data-floor="${active}"><circle class="tip" cx="${cxm}" cy="${y5 - b - 38}" r="4"/></g>`;
  return s;
}
